% Academic CV: a curriculum vitae written as a Markdown document.
% From grasmere.app/markdown/academic-cv-template.html. Free to use and change.
%
% The template has Pandoc's slots, so it works two ways.
%   With Pandoc:    pandoc cv.md --shift-heading-level-by=-1 --template academic-cv.tex -o cv.pdf
%   With Grasmere:  keep this file in your project, then choose it under Publish, LaTeX.
% Your name is the author and the title sits beneath it. The text goes where the body slot is.
\documentclass[11pt]{article}
% The type is Palatino. pdfLaTeX, XeLaTeX and LuaLaTeX all work.
\usepackage{iftex}
\ifPDFTeX
  \usepackage[utf8]{inputenc}
  \usepackage[T1]{fontenc}
  \usepackage[sc]{mathpazo}
\else
  \usepackage{fontspec}
  \setmainfont{texgyrepagella}[Extension=.otf,UprightFont=*-regular,BoldFont=*-bold,ItalicFont=*-italic,BoldItalicFont=*-bolditalic]
\fi
\linespread{1.05}
% Paper and margins. Write letterpaper for US Letter.
\usepackage[a4paper,margin=1in]{geometry}
\usepackage{graphicx}
\usepackage{titlesec}
\usepackage{enumitem}
\usepackage{fancyhdr}
\usepackage{longtable,booktabs,array,calc}
\usepackage[hidelinks]{hyperref}

% Paragraphs: no indent, a little space between, ragged on the right.
\setlength{\parindent}{0pt}
\setlength{\parskip}{0.5em}
\raggedright

% Headings. A section is small capitals over a rule; the two below it are bold and italic.
\setcounter{secnumdepth}{0}
\titleformat{\section}{\large\scshape\raggedright}{}{0pt}{}[\vspace{0.15em}\titlerule]
\titlespacing*{\section}{0pt}{1.5em}{0.5em}
\titleformat{\subsection}{\normalsize\bfseries\raggedright}{}{0pt}{}
\titlespacing*{\subsection}{0pt}{0.9em}{0.1em}
\titleformat{\subsubsection}{\normalsize\itshape\raggedright}{}{0pt}{}
\titlespacing*{\subsubsection}{0pt}{0.7em}{0.1em}

% Lists. The first level has no bullets and a hanging indent, as a list of publications
% is usually set. A list inside a list has bullets.
\setlist[itemize,1]{label={},labelwidth=0pt,labelsep=0pt,leftmargin=1.5em,itemindent=-1.5em,itemsep=0.35em,parsep=0pt,topsep=0.2em}
\setlist[itemize,2]{label=\textbullet,labelwidth=0.6em,labelsep=0.5em,leftmargin=1.1em,itemindent=0pt,itemsep=0.1em,parsep=0pt,topsep=0.1em}
\setlist[enumerate]{leftmargin=1.5em,itemsep=0.35em,parsep=0pt,topsep=0.2em}

% The foot of each page: your name, the date, the page number.
\pagestyle{fancy}
\fancyhf{}
\renewcommand{\headrulewidth}{0pt}
\fancyfoot[L]{\footnotesize $if(author)$$author$$else$$title$$endif$}
$if(date)$
\fancyfoot[C]{\footnotesize $date$}
$endif$
\fancyfoot[R]{\footnotesize\thepage}

% What Pandoc's LaTeX expects to find. Grasmere doesn't use these.
\providecommand{\tightlist}{}
\providecommand{\pandocbounded}[1]{#1}

\begin{document}
$if(author)$
{\huge\scshape $author$}\par
$if(title)$
\vspace{0.1em}{\large\itshape $title$}\par
$endif$
$else$
{\huge\scshape $title$}\par
$endif$
\vspace{0.5em}\hrule\vspace{0.9em}

$body$

\end{document}
